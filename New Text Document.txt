
#include <stdio.h>
#include <string.h>

#define MAX 50

int main()
{
    char name[MAX][30];
    int runs[MAX];
    int balls[MAX];
    int count = 0;
    int choice = 0, i, found;
    char key[30];
    float sr;

    do {
        printf("\n===== CRICKET PLAYER STATS TRACKER =====\n");
        printf("1. Add Player\n");
        printf("2. View All Players\n");
        printf("3. Search Player by Name\n");
        printf("4. Exit\n");
        printf("Enter your choice: ");
        scanf("%d", &choice);

        switch (choice) {

        case 1:
            if (count == MAX) {
                printf("Cannot add: Storage full.\n");
                break;
            }

            printf("Enter player name: ");
            scanf(" %29[^\n]", name[count]);

            do {
                printf("Enter runs scored: ");
                scanf("%d", &runs[count]);

                if (runs[count] < 0) {
                    printf("Invalid input! Runs cannot be negative.\n");
                }

            } while (runs[count] < 0);

            do {
                printf("Enter balls faced: ");
                scanf("%d", &balls[count]);

                if (balls[count] < 0) {
                    printf("Invalid input! Balls cannot be negative.\n");
                }
                else if (balls[count] == 0 && runs[count] > 0) {
                    printf("Invalid input! A player cannot score runs without facing a ball.\n");
                }
                else if (balls[count] > 0 && runs[count] > balls[count] * 6) {
                    printf("Invalid input! Runs cannot exceed 6 times the balls faced.\n");
                }

            } while (balls[count] < 0 ||
                     (balls[count] == 0 && runs[count] > 0) ||
                     (balls[count] > 0 && runs[count] > balls[count] * 6));

            count++;

            printf("Player added successfully with S.No. %d\n", count);
            break;

        case 2:
            if (count == 0) {
                printf("No players added yet.\n");
                break;
            }

            printf("\n%-6s %-25s %6s %6s %12s\n",
                   "S.No.", "Name", "Runs", "Balls", "Strike Rate");

            printf("----------------------------------------------------------\n");

            for (i = 0; i < count; i++) {

                if (balls[i] > 0)
                    sr = (runs[i] * 100.0) / balls[i];
                else
                    sr = 0;

                printf("%-6d %-25s %6d %6d %12.2f\n",
                       i + 1, name[i], runs[i], balls[i], sr);
            }
            break;

        case 3:
            if (count == 0) {
                printf("No players added yet.\n");
                break;
            }

            printf("Enter player name to search: ");
            scanf(" %29[^\n]", key);

            found = 0;

            for (i = 0; i < count; i++) {

                if (strcmp(name[i], key) == 0) {

                    if (balls[i] > 0)
                        sr = (runs[i] * 100.0) / balls[i];
                    else
                        sr = 0;

                    printf("\nS.No.: %d\n", i + 1);
                    printf("Name: %s\n", name[i]);
                    printf("Runs: %d\n", runs[i]);
                    printf("Balls: %d\n", balls[i]);
                    printf("Strike Rate: %.2f\n", sr);

                    found = 1;
                }
            }

            if (found == 0)
                printf("No player found with name %s\n", key);

            break;

        case 4:
            printf("Exiting. Goodbye!\n");
            break;

        default:
            printf("Invalid choice. Enter 1 to 4.\n");
        }

    } while (choice != 4);

    return 0;
}
